cmake_minimum_required(VERSION 3.19)
project(SearchAdvisor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(STS_REFERENCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../sts-rl-agent-principles" CACHE PATH "Read-only reference checkout")
set(SRC "${STS_REFERENCE_ROOT}/combat_engine/combat4r/engine-source")
if(NOT EXISTS "${SRC}/src/combat/BattleContext.cpp")
  message(FATAL_ERROR "Set STS_REFERENCE_ROOT to sts-rl-agent-principles")
endif()
file(GLOB_RECURSE CORE_SOURCES CONFIGURE_DEPENDS "${SRC}/src/*.cpp")
list(FILTER CORE_SOURCES EXCLUDE REGEX "/BattleScumSearcher2.cpp$")
add_library(sts_core STATIC ${CORE_SOURCES} vendor/BattleScumSearcher2.cpp)
target_include_directories(sts_core PUBLIC "${SRC}/include" "${SRC}/bindings" "${STS_REFERENCE_ROOT}/combat_engine/third_party" "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_definitions(sts_core PUBLIC COMBAT3_TARGET_POLICY=1)
if(MSVC)
  target_compile_options(sts_core PRIVATE /O2 /UNDEBUG /bigobj /utf-8)
  target_compile_definitions(sts_core PUBLIC NOMINMAX _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(sts_core PRIVATE -O2 -UNDEBUG)
endif()
find_package(Threads REQUIRED)
add_executable(sts-search-server src/server.cpp src/adapter.cpp)
target_include_directories(sts-search-server PRIVATE vendor)
target_link_libraries(sts-search-server PRIVATE sts_core Threads::Threads)
if(NOT MSVC)
  target_compile_options(sts-search-server PRIVATE -Wall -Wextra -UNDEBUG)
endif()
install(TARGETS sts-search-server RUNTIME DESTINATION bin)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  enable_testing()
  add_test(NAME protocol_and_fixtures COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../tests/test_server.py" --server "$<TARGET_FILE:sts-search-server>" --reference "${STS_REFERENCE_ROOT}")
  set_tests_properties(protocol_and_fixtures PROPERTIES TIMEOUT 180)
endif()
